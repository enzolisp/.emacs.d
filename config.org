#+begin_src emacs-lisp
  (defun tangle-init ()
    "If the current buffer is init.org the code-blocks are
  tangled, and the tangled file is compiled."
    (when (equal (buffer-file-name)
                 (expand-file-name (concat user-emacs-directory "config.org")))
      ;; Avoid running hooks when tangling.
      (let ((prog-mode-hook nil))
        (org-babel-tangle)
        (byte-compile-file (concat user-emacs-directory "init.el")))))

  (add-hook 'after-save-hook 'tangle-init)


  (defun my-reload-config ()
    (interactive)
    (org-babel-load-file "~/.emacs.d/config.org"))

  (defun indent-region-or-buffer ()
    (interactive)
    (save-excursion
      (indent-region (point-min) (point-max))))

  (defun my-backspace ()
    (interactive)
    (if (looking-back "^[[:space:]]+" (line-beginning-position))
        (indent-rigidly (line-beginning-position) (point) (- tab-width))
      (delete-char -1)))

  (defun my-move-end-of-line ()        ; C-e moves the cursor to before the first char of the line +
    (interactive)                      ; if it is before the first char of the line
    (if (< (point) (progn (back-to-indentation) (point)))
        (back-to-indentation)
      (end-of-line)))

  (defun my-move-beginning-of-line ()  ; C-a moves the cursor  before the first char of the line. 
    (interactive)                      ; pressing again moves to the beginning of the line
    (if (= (point) (progn (back-to-indentation) (point)))
        (beginning-of-line)))

  (defun my-pre-compile-command (&rest args)
    (setq file_name (file-name-nondirectory (or (buffer-file-name) (buffer-name))))
    (setq file_name_sans_ext (file-name-base (file-name-sans-extension (or (buffer-file-name) (buffer-name))))))


  (advice-add 'compile :before #'my-pre-compile-command)

  (defun my/kill-line-no-ring ()
    "Kill line; if line is blank, delete without adding to kill ring."
    (interactive)
    (if (string-blank-p (thing-at-point 'line t))
        (delete-region (line-beginning-position) (min (1+ (line-end-position)) (point-max)))
      (kill-line)))


#+end_src
